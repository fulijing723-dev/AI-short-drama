];
  }

  function writablePathParts(key) {
    const parts = readPathParts(key);
    const [topLevel] = parts;
    if (!WRITABLE_STATE_KEYS.has(topLevel)) throw new TypeError(`State field is not writable: ${topLevel || key}`);
    if ((topLevel === 'selectedTab' || topLevel === 'drawer') && parts.length !== 1) {
      throw new TypeError(`State field does not accept nested keys: ${topLevel}`);
    }
    if ((topLevel === 'selections' || topLevel === 'inputs') && (parts.length < 1 || parts.length > 2)) {
      throw new TypeError(`State record path must contain at most one nested key: ${key}`);
    }
    return parts;
  }

  function getLocalState(key, fallback) {
    const parts = readPathParts(key);
    if (parts.length === 0) return fallback;
    let value = state;
    for (const part of parts) {
      if (!value || typeof value !== 'object' || !Object.hasOwn(value, part)) return fallback;
      value = value[part];
    }
    return value;
  }

  function setLocalState(key, value) {
    const parts = writablePathParts(key);
    const [topLevel, nestedKey] = parts;
    const candidate = {
      screen: state.screen,
      selectedTab: state.selectedTab,
      drawer: state.drawer,
      selections: normalizeStringRecord(state.selections),
      inputs: normalizeStringRecord(state.inputs),
    };

    if (topLevel === 'selectedTab') {
      if (typeof value !== 'string') throw new TypeError('selectedTab must be a string');
      candidate.selectedTab = value;
    } else if (topLevel === 'drawer') {
      if (typeof value !== 'string' && value !== null) throw new TypeError('drawer must be a string or null');
      candidate.drawer = value;
    } else if (nestedKey) {
      if (typeof value !== 'string') throw new TypeError(`${topLevel}.${nestedKey} must be a string`);
      candidate[topLevel][nestedKey] = value;
    } else {
      candidate[topLevel] = normalizeStringRecord(value);
    }

    const serialized = JSON.stringify(candidate);
    window.localStorage.setItem(STORAGE_KEY, serialized);
    state.screen = candidate.screen;
    state.selectedTab = candidate.selectedTab;
    state.drawer = candidate.drawer;
    state.selections = candidate.selections;
    state.inputs = candidate.inputs;
    window.dispatchEvent(new CustomEvent('shortdrama:statechange', { detail: { key, value, state } }));
    return value;
  }

  function registerScreen(id, render) {
    if (typeof id !== 'string' || id.length === 0) throw new TypeError('Screen ID must be a non-empty string');
    if (typeof render !== 'function') throw new TypeError(`Screen renderer must be a function: ${id}`);
    registry.set(id, render);
    return render;
  }

  function idSegment(value) {
    return String(value || 'control').toLowerCase().replace(/[^a-z0-9_-]+/g, '-').replace(/^-+|-+$/g, '') || 'control';
  }

  function ensureStableNavigationId(control, screenId = state.screen) {
    if (control.id) return control.id;
    const target = control.dataset.nav || 'route';
    const peers = [...app.querySelectorAll('[data-nav]')].filter((item) => item.dataset.nav === target);
    const position = Math.max(0, peers.indexOf(control));
    control.id = `shortdrama-nav-${idSegment(screenId)}-${idSegment(target)}-${position + 1}`;
    return control.id;
  }

  function assignStableNavigationIds(screenId) {
    for (const control of app.querySelectorAll('[data-nav]')) ensureStableNavigationId(control, screenId);
  }

  function focusableControls(container) {
    return [...container.querySelectorAll(FOCUSABLE_SELECTOR)].filter((control) => {
      if (control.closest('[inert], [aria-hidden="true"]')) return false;
      const style = window.getComputedStyle(control);
      return style.display !== 'none' && style.visibility !== 'hidden';
    });
  }

  function focusDialog(dialog) {
    const preferred = dialog.querySelector('[data-modal-initial-focus]') || focusableControls(dialog)[0];
    const target = preferred || dialog;
    if (!preferred && !dialog.hasAttribute('tabindex')) dialog.tabIndex = -1;
    target.focus({ preventScroll: true });
  }

  function restoreStateControlFocus(screenId) {
    if (!pendingStateFocus || pendingStateFocus.screenId !== screenId) return false;
    const focusState = pendingStateFocus;
    const match = [...app.querySelectorAll('[data-state-key]')].find((control) => (
      control.dataset.stateKey === focusState.key
      && (control.dataset.stateValue ?? control.value) === focusState.value
    ));
    pendingStateFocus = null;
    if (!match) return false;
    match.focus({ preventScroll: true });
    if (
      Number.isInteger(focusState.selectionStart)
      && Number.isInteger(focusState.selectionEnd)
      && typeof match.setSelectionRange === 'function'
    ) {
      match.setSelectionRange(focusState.selectionStart, focusState.selectionEnd, focusState.selectionDirection || 'none');
    }
    return true;
  }

  function restoreNavigationFocus() {
    if (!pendingFocusRestoreId) return false;
    const target = document.getElementById(pendingFocusRestoreId);
    pendingFocusRestoreId = '';
    if (!target || !app.contains(target)) return false;
    target.focus({ preventScroll: true });
    return true;
  }

  function finishRenderedAccessibility(screenId) {
    assignStableNavigationIds(screenId);
    const restoredStateFocus = restoreStateControlFocus(screenId);
    const dialog = app.querySelector(MODAL_SELECTOR);
    if (dialog) {
      if (pendingNavigationOrigin?.targetScreen === screenId) activeModalOrigin = pendingNavigationOrigin;
      pendingNavigationOrigin = null;
      if (!restoredStateFocus) {
        focusDialog(dialog);
        queueMicrotask(() => {
          if (app.contains(dialog) && !dialog.contains(document.activeElement)) focusDialog(dialog);
        });
      }
      return;
    }
    pendingNavigationOrigin = null;
    if (!restoredStateFocus) restoreNavigationFocus();
  }

  function prepareModalClose() {
    if (activeModalOrigin?.triggerId) pendingFocusRestoreId = activeModalOrigin.triggerId;
    activeModalOrigin = null;
  }

  function updateUrl(screenId, replace) {
    const url = new URL(window.location.href);
    url.searchParams.set('screen', screenId);
    const nextIndex = replace ? currentHistoryIndex : currentHistoryIndex + 1;
    const historyState = {
      ...(window.history.state && typeof window.history.state === 'object' ? window.history.state : {}),
      screen: screenId,
      [HISTORY_INDEX_KEY]: nextIndex,
    };
    window.history[replace ? 'replaceState' : 'pushState'](historyState, '', url);
    currentHistoryIndex = nextIndex;
  }

  function renderScreen(screenId) {
    const render = registry.get(screenId);
    if (!render) throw new Error(`No renderer registered for screen: ${screenId}`);
    state.screen = screenId;
    persistState();
    app.dataset.screen = screenId;
    document.body.dataset.screen = screenId;
    app.innerHTML = render(state);
    const screenRoot = app.firstElementChild;
    if (!screenRoot) throw new Error(`Screen renderer returned no root element: ${screenId}`);
    screenRoot.dataset.screenId = screenId;
    finishRenderedAccessibility(screenId);
  }

  function navigate(id, options = { replace: false }) {
    const requestedId = typeof id === 'string' && id ? id : 'home';
    const knownRoute = registry.has(requestedId);
    const screenId = knownRoute ? requestedId : 'home';
    const replace = Boolean(options?.replace) || !knownRoute;
    updateUrl(screenId, replace);
    renderScreen(screenId);
    return screenId;
  }

  function back(fallbackId = 'home') {
    if (currentHistoryIndex > 0) {
      window.history.back();
      return;
    }
    navigate(fallbackId, { replace: true });
  }

  function handleRouteFromHistory(event) {
    const targetIndex = event.state?.[HISTORY_INDEX_KEY];
    if (!isHistoryIndex(targetIndex)) {
      navigate('home', { replace: true });
      return;
    }
    const direction = Math.sign(targetIndex - currentHistoryIndex);
    currentHistoryIndex = targetIndex;
    const requestedId = new URLSearchParams(window.location.search).get('screen') || 'home';
    if (!registry.has(requestedId)) {
      navigate('home', { replace: true });
      return;
    }
    if (direction < 0 && app.querySelector(MODAL_SELECTOR)) prepareModalClose();
    renderScreen(requestedId);
    window.dispatchEvent(new CustomEvent('shortdrama:historychange', {
      detail: { direction, index: currentHistoryIndex, screen: requestedId },
    }));
  }

  function delegatedControl(event) {
    if (!(event.target instanceof Element)) return null;
    const control = event.target.closest('[data-nav], [data-back], [data-state-key], [data-action]');
    return control && control !== app && app.contains(control) ? control : null;
  }

  app.addEventListener('click', (event) => {
    const control = delegatedControl(event);
    if (!control) return;

    if (control.dataset.nav) {
      event.preventDefault();
      if (control.dataset.beforeStateKey) {
        setLocalState(control.dataset.beforeStateKey, control.dataset.beforeStateValue || '');
      } else if (control.dataset.nav === 'detail') {
        setLocalState('selections.selected_drama', 'village');
      }
      pendingNavigationOrigin = {
        sourceScreen: state.screen,
        targetScreen: control.dataset.nav,
        triggerId: ensureStableNavigationId(control),
      };
      navigate(control.dataset.nav);
      return;
    }
    if (Object.hasOwn(control.dataset, 'back')) {
      event.preventDefault();
      if (control.closest(MODAL_SELECTOR)) prepareModalClose();
      back(control.dataset.back || 'home');
      return;
    }
    if (control.dataset.stateKey) {
      const value = control.dataset.stateValue ?? control.value;
      pendingStateFocus = { screenId: state.screen, key: control.dataset.stateKey, value };
      setLocalState(control.dataset.stateKey, value);
      if (control.isConnected) pendingStateFocus = null;
    }
    if (control.dataset.action === 'drawer-open') {
      setLocalState('drawer', control.dataset.drawer || control.dataset.stateValue || null);
    } else if (control.dataset.action === 'drawer-close') {
      setLocalState('drawer', null);
    } else if (control.dataset.action === 'tab-select') {
      setLocalState('selectedTab', control.dataset.tab || control.dataset.stateValue || '');
    } else if (control.dataset.action === 'toast') {
      showToast(control.dataset.message || '操作已完成');
    }
  });

  app.addEventListener('input', (event) => {
    const control = event.target instanceof Element ? event.target.closest('[data-state-key]') : null;
    if (!control || control === app || !app.contains(control)) return;
    const value = control.dataset.stateValue ?? control.value;
    pendingStateFocus = {
      screenId: state.screen,
      key: control.dataset.stateKey,
      value,
      selectionStart: Number.isInteger(control.selectionStart) ? control.selectionStart : null,
      selectionEnd: Number.isInteger(control.selectionEnd) ? control.selectionEnd : null,
      selectionDirection: control.selectionDirection || 'none',
    };
    setLocalState(control.dataset.stateKey, value);
    if (control.isConnected) pendingStateFocus = null;
  });

  app.addEventListener('keydown', (event) => {
    const dialog = app.querySelector(MODAL_SELECTOR);
    if (!dialog) return;
    if (event.key === 'Escape') {
      const closeControl = dialog.querySelector('[data-back]');
      if (!closeControl) return;
      event.preventDefault();
      event.stopImmediatePropagation();
      prepareModalClose();
      navigate(closeControl.dataset.nav || closeControl.dataset.back || 'home', { replace: true });
      return;
    }
    if (event.key !== 'Tab') return;
    const controls = focusableControls(dialog);
    if (controls.length === 0) {
      event.preventDefault();
      focusDialog(dialog);
      return;
    }
    const first = controls[0];
    const last = controls[controls.length - 1];
    const active = document.activeElement;
    if (!dialog.contains(active) || (event.shiftKey && active === first) || (!event.shiftKey && active === last)) {
      event.preventDefault();
      (event.shiftKey ? last : first).focus({ preventScroll: true });
    }
  });

  window.addEventListener('popstate', handleRouteFromHistory);
  window.addEventListener('DOMContentLoaded', () => {
    const requestedId = new URLSearchParams(window.location.search).get('screen') || 'home';
    navigate(requestedId, { replace: true });
  }, { once: true });

  const api = {
    state,
    registerScreen,
    navigate,
    back,
    setLocalState,
    getLocalState,
    getRegisteredScreenIds: () => [...registry.keys()],
  };
  window.ShortDramaApp = api;
  window.registerScreen = registerScreen;
  window.navigate = navigate;
  window.back = back;
  window.setLocalState = setLocalState;
  window.getLocalState = getLocalState;
  window.showToast = showToast;
})();

    if (!vote || !app.contains(vote)) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    const key = vote.dataset.wishVoteKey;
    const alreadyVoted = window.getLocalState(`selections.${key}_date`, '') === wishTodayKey();
    if (alreadyVoted) {
      window.setLocalState('selections.wish_vote_prompt', 'already');

      window.setLocalState('selections.wish_vote_prompt', 'success');
    }
    window.navigate(app.dataset.screen, { replace: true });
  }, true);
  document.addEventListener('keydown', (event) => {
    if (event.key !== 'Escape' || !app.querySelector('.wish-prompt')) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    window.setLocalState('selections.wish_vote_prompt', '');
    window.navigate(app.dataset.screen, { replace: true });
  }, true);
  app.addEventListener('keydown', (event) => {
    if (!['Enter', ' '].includes(event.key)) return;
    const control = event.target instanceof Element ? event.target.closest('[data-nav], [data-back]') : null;
    if (!control || !app.contains(control)) return;
    event.preventDefault();
    control.click();
  });

      window.navigate(app.dataset.screen, { replace: true });
    }
  });

  window.ShortDramaBrowseScreens = Object.freeze({ ...routes });
})();

  function enhanceRenewalEntrypoints() {
    if (app.dataset.screen !== 'profile') return;
    const chaseCell = app.querySelector('.chase-cell');
    if (!chaseCell) return;
    chaseCell.dataset.nav = 'renewal-center';
    chaseCell.setAttribute('role', 'button');
    chaseCell.setAttribute('aria-label', '催更中心，催更想看的，投票增加更');

  new MutationObserver(enhanceRenewalEntrypoints).observe(app, { childList: true });

  document.addEventListener('keydown', (event) => {
    if (event.key !== 'Escape' || app.dataset.screen !== 'nominate-missing' || !app.querySelector('.nomination-panel')) return;
    event.preventDefault();
    event.stopImmediatePropagation();
    window.navigate('wishlist');
  }, true);

  app.addEventListener('click', (event) => {
    if (!(event.target instanceof Element)) return;
    const tag = event.target.closest('[data-nomination-tag]');
    if (tag && app.contains(tag)) {
      const selected = tag.getAttribute('aria-pressed') === 'true';
      const selectedCount = app.querySelectorAll('.nomination-tag[aria-pressed="true"]').length;
      if (!selected && selectedCount >= 3) return;
      tag.setAttribute('aria-pressed', String(!selected));
      tag.classList.toggle('selected', !selected);
      return;
    }

    const renewalVote = event.target.closest('[data-renewal-vote]');
    if (renewalVote && app.contains(renewalVote)) {
      const index = Number(renewalVote.dataset.renewalIndex);
      const key = `selections.renewal_vote_${index}`;
      if (window.getLocalState(key, '') === 'voted') {
        window.setLocalState('selections.renewal_prompt', 'already');
      } else if (renewalTicketCount() <= 0) {
        window.setLocalState('selections.renewal_prompt', 'insufficient');
      } else {
        window.setLocalState(key, 'voted');
        window.setLocalState('selections.renewal_prompt', 'success');
      }
      window.navigate('renewal-center', { replace: true });
      return;
    }

    const promptClose = event.target.closest('[data-renewal-prompt-close]');
    if (promptClose && app.contains(promptClose)) {
      window.setLocalState('selections.renewal_prompt', '');
      window.navigate('renewal-center', { replace: true });
      return;
    }

    const renewalDay = event.target.closest('[data-renewal-day]');
    if (renewalDay && app.contains(renewalDay)) {
      if (renewalDay.disabled || renewalDay.dataset.dayIndex !== '0') return;
      window.setLocalState('selections.renewal_checkin', 'signed');
      window.navigate('renewal-center', { replace: true });
      return;
    }

    const submit = event.target.closest('[data-nomination-submit]');
    if (submit && app.contains(submit)) {
      if (app.dataset.screen === 'nominate') {
        window.showToast('请先选择一部短剧，再选择魔改方向');
        app.querySelector('.nomination-drama')?.focus();
        return;
      }
      const direction = window.getLocalState('selections.remix_direction', '');
      if (app.dataset.screen === 'remix-direction') {
        if (!direction) {
          window.showToast('请选择一个魔改方向');
          app.querySelector('.remix-direction-option')?.focus();
          return;
        }
        window.setLocalState('selections.nomination_submitted', 'submitted');
        window.showToast('提名已提交，感谢你的创意');
        window.setTimeout(() => window.navigate('wishlist'), 300);
        return;
      }
      if (app.dataset.screen === 'nominate-missing') {
        const title = app.querySelector('#missing-title')?.value.trim() || '';
        if (!title) {
          window.showToast('请输入短剧名称');
          app.querySelector('#missing-title')?.focus();
          return;
        }
        if (!direction) {
          window.showToast('请选择一个魔改方向');
          app.querySelector('.missing-direction-select')?.focus();
          return;
        }
        window.setLocalState('selections.nomination_submitted', 'submitted');
        window.showToast('未收录短剧提名已提交');
        window.setTimeout(() => window.navigate('wishlist'), 300);
      }
    }
  });

  app.addEventListener('change', (event) => {
    const input = event.target instanceof Element ? event.target.closest('#missing-cover') : null;
    if (!input || !app.contains(input)) return;
    const label = app.querySelector('.nomination-upload-label');
    const fileName = input.files?.[0]?.name || '';
    if (label) label.textContent = fileName || '上传封面';
    if (fileName) window.showToast(`已在本地选择 ${fileName}`);
  });

  app.addEventListener('keydown', (event) => {
    if (event.key === 'Escape' && app.querySelector('.renewal-prompt')) {
      event.preventDefault();
      event.stopImmediatePropagation();
      window.setLocalState('selections.renewal_prompt', '');
      window.navigate('renewal-center', { replace: true });
      return;
    }
    if (!['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight'].includes(event.key)) return;
    const radio = event.target instanceof Element ? event.target.closest('.remix-direction-option[role="radio"]') : null;
    if (!radio || !app.contains(radio)) return;
    const radios = [...app.querySelectorAll('.remix-direction-option[role="radio"]')];
    const currentIndex = radios.indexOf(radio);
    if (currentIndex < 0) return;
    event.preventDefault();
    const delta = ['ArrowDown', 'ArrowRight'].includes(event.key) ? 1 : -1;
    const next = radios[(currentIndex + delta + radios.length) % radios.length];
    next.focus({ preventScroll: true });
    next.click();
  });

  window.addEventListener('shortdrama:statechange', (event) => {
    const key = String(event.detail?.key || '');
    if (key === 'inputs.nomination_query' && app.dataset.screen === 'nominate') {
      window.navigate('nominate', { replace: true });
      return;
    }
    if (key === 'selections.remix_direction' && ['remix-direction', 'nominate-missing'].includes(app.dataset.screen)) {
      const selected = String(event.detail?.value || '');
      for (const option of app.querySelectorAll('.remix-direction-option')) {
        const active = option.dataset.stateValue === selected;
        option.setAttribute('aria-checked', String(active));
        option.tabIndex = active ? 0 : -1;
        option.classList.toggle('selected', active);
      }
      const display = app.querySelector('.direction-select-display');
      if (display) display.firstChild.textContent = selected || '选择方向';
      const select = app.querySelector('.missing-direction-select');
      if (select) select.value = selected;
    }
    if (key === 'selections.renewal_prompt' && app.dataset.screen === 'renewal-center') {
      window.navigate('renewal-center', { replace: true });
    }
  });

  window.ShortDramaNominationScreens = Object.freeze({ ...routes });
})();
